-- ============================================
-- MM2 Position Copier
-- Tap anywhere — copies coordinates to clipboard
-- ============================================

local Players = game:GetService("Players")
local UIS = game:GetService("UserInputService")
local LocalPlayer = Players.LocalPlayer
local PlayerGui = LocalPlayer:WaitForChild("PlayerGui")

-- build GUI
local screen = Instance.new("ScreenGui")
screen.Name = "PosCopier"
screen.ResetOnSpawn = false
screen.Parent = PlayerGui

local frame = Instance.new("Frame")
frame.Size = UDim2.new(1, 0, 1, 0)
frame.Position = UDim2.new(0, 0, 0, 0)
frame.BackgroundTransparency = 1
frame.Parent = screen

local label = Instance.new("TextLabel")
label.Size = UDim2.new(0, 300, 0, 50)
label.Position = UDim2.new(0.5, -150, 0, 20)
label.BackgroundColor3 = Color3.fromRGB(30, 30, 30)
label.TextColor3 = Color3.fromRGB(0, 255, 100)
label.Text = "TAP ACCEPT BUTTON TO COPY COORDS"
label.TextScaled = true
label.Font = Enum.Font.GothamBold
label.Parent = frame

-- capture any touch/click
UIS.InputBegan:Connect(function(input, gpe)
    if input.UserInputType == Enum.UserInputType.Touch
    or input.UserInputType == Enum.UserInputType.MouseButton1 then
        local x = math.floor(input.Position.X)
        local y = math.floor(input.Position.Y)
        local coords = "X=" .. x .. " Y=" .. y
        label.Text = coords
        setclipboard(coords)
        print("[PosCopier] Copied: " .. coords)
    end
end)

print("[PosCopier] Running — tap the accept button now")
